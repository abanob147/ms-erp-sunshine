-- Sunshine ERP: مندوب التحصيل واعتماد استلام الشركة -- شغّل هذا الملف مرة واحدة من Supabase Dashboard > SQL Editor. -- يعتمد على الجداول/الدوال الموجودة بالفعل في النظام: -- sales_reps, sales_rep_transactions, sales, current_user_role(), -- record_sales_rep_customer_receipt(...) begin; create table if not exists public.sales_rep_collection_receipts ( id bigint generated by default as identity primary key, sales_rep_id bigint not null, customer_id bigint not null, amount numeric(14,2) not null check (amount > 0), receipt_date date not null default current_date, notes text, status text not null default 'pending' check (status in ('pending','approved')), created_by uuid not null default auth.uid(), approved_by uuid, approved_at timestamptz, created_at timestamptz not null default now() ); create index if not exists sales_rep_collection_receipts_status_date_idx on public.sales_rep_collection_receipts(status, receipt_date desc); create index if not exists sales_rep_collection_receipts_rep_idx on public.sales_rep_collection_receipts(sales_rep_id, created_at desc); alter table public.sales_rep_collection_receipts enable row level security; drop policy if exists sales_rep_collection_receipts_select on public.sales_rep_collection_receipts; create policy sales_rep_collection_receipts_select on public.sales_rep_collection_receipts for select to authenticated using ( public.current_user_role() = 'admin' or (public.current_user_role() = 'employee' and not exists ( select 1 from public.sales_reps own_rep where own_rep.user_id = auth.uid() and own_rep.is_active = true )) or exists ( select 1 from public.sales_reps r where r.id = sales_rep_collection_receipts.sales_rep_id and r.user_id = auth.uid() and r.is_active = true ) ); -- لا نسمح بإدخال/تعديل مباشر من المتصفح؛ كل العمليات تمر عبر الدوال أدناه. revoke insert, update, delete on public.sales_rep_collection_receipts from anon, authenticated; grant select on public.sales_rep_collection_receipts to authenticated; create or replace function public.submit_sales_rep_collection_receipt( p_customer_id bigint, p_amount numeric, p_date date default current_date, p_notes text default null ) returns bigint language plpgsql security definer set search_path = public as $$ declare v_rep public.sales_reps%rowtype; v_id bigint; begin if auth.uid() is null then raise exception 'يجب تسجيل الدخول أولاً'; end if; if p_customer_id is null or p_amount is null or p_amount <= 0 then raise exception 'بيانات التحصيل غير صحيحة'; end if; select * into v_rep from public.sales_reps where user_id = auth.uid() and is_active = true limit 1; if not found then raise exception 'هذا الحساب غير مرتبط بمندوب نشط'; end if; -- لا يُسجل المندوب تحصيلاً إلا لعميل سبق أن باعه من خلال هذا الحساب. if not exists ( select 1 from public.sales_rep_transactions rt join public.sales s on s.id = rt.sale_id where rt.sales_rep_id = v_rep.id and s.customer_id = p_customer_id ) then raise exception 'هذا العميل ليس ضمن مبيعات هذا المندوب'; end if; insert into public.sales_rep_collection_receipts (sales_rep_id, customer_id, amount, receipt_date, notes, status, created_by) values (v_rep.id, p_customer_id, round(p_amount,2), coalesce(p_date,current_date), nullif(trim(p_notes),''), 'pending', auth.uid()) returning id into v_id; return v_id; end; $$; create or replace function public.approve_sales_rep_collection_receipt( p_receipt_id bigint ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_receipt public.sales_rep_collection_receipts%rowtype; v_role text; v_description text; begin if auth.uid() is null then raise exception 'يجب تسجيل الدخول أولاً'; end if; v_role := public.current_user_role(); if v_role not in ('admin','employee') then raise exception 'غير مصرح باعتماد التحصيل'; end if; -- حساب المندوب لا يمكنه اعتماد تحصيله بنفسه. if exists (select 1 from public.sales_reps r where r.user_id = auth.uid() and r.is_active = true) then raise exception 'اعتماد التحصيل متاح لموظفي الشركة فقط'; end if; select * into v_receipt from public.sales_rep_collection_receipts where id = p_receipt_id for update; if not found then raise exception 'سند التحصيل غير موجود'; end if; if v_receipt.status = 'approved' then raise exception 'تم اعتماد هذا التحصيل من قبل'; end if; v_description := 'تحصيل مندوب - سند رقم ' || v_receipt.id; if nullif(trim(v_receipt.notes),'') is not null then v_description := v_description || ' - ' || trim(v_receipt.notes); end if; -- هذه الدالة الموجودة في النظام هي التي ترحّل القبض إلى كشف العميل -- وحساب المندوب والخزينة. لا يتم استدعاؤها عند إنشاء السند المعلّق. perform public.record_sales_rep_customer_receipt( p_sales_rep_id => v_receipt.sales_rep_id, p_customer_id => v_receipt.customer_id, p_amount => v_receipt.amount, p_date => v_receipt.receipt_date, p_description => v_description, p_account_type => 'cash', p_account_id => null ); update public.sales_rep_collection_receipts set status = 'approved', approved_by = auth.uid(), approved_at = now() where id = v_receipt.id; return jsonb_build_object('success', true, 'receipt_id', v_receipt.id, 'status', 'approved'); end; $$; revoke all on function public.submit_sales_rep_collection_receipt(bigint,numeric,date,text) from public, anon; revoke all on function public.approve_sales_rep_collection_receipt(bigint) from public, anon; grant execute on function public.submit_sales_rep_collection_receipt(bigint,numeric,date,text) to authenticated; grant execute on function public.approve_sales_rep_collection_receipt(bigint) to authenticated; commit;
